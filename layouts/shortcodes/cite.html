{{- $keysString := .Get 0 -}} {{- $style := .Get 1 -}} {{- $page := .Page -}} {{- /* Split the keys string by space into an array */ -}} {{- $keys := split $keysString " " -}} {{- /* Initialize an array to hold the rendered inline HTML blocks */ -}} {{- $renderedCites := slice -}} {{- range $key := $keys -}} {{- /* Check BOTH 'references' and 'reference' just in case */ -}} {{- $dataSource := $.Site.Data.references -}} {{- if not $dataSource -}} {{- $dataSource = $.Site.Data.reference -}} {{- end -}} {{- $entry := index (where $dataSource "id" $key) 0 -}} {{- if $entry -}} {{- $authorText := "Unknown" -}} {{- if $entry.author -}} {{- $firstAuthor := index $entry.author 0 -}} {{- $count := len $entry.author -}} {{- if eq $count 1 -}} {{- $authorText = $firstAuthor.family -}} {{- else if eq $count 2 -}} {{- $secondAuthor := index $entry.author 1 -}} {{- $authorText = printf "%s & %s" $firstAuthor.family $secondAuthor.family -}} {{- else -}} {{- $authorText = printf "%s et al." $firstAuthor.family -}} {{- end -}} {{- end -}} {{- $year := "n.d." -}} {{- if index $entry "issued" -}} {{- $dateParts := index (index $entry "issued" "date-parts") 0 -}} {{- $year = index $dateParts 0 -}} {{- end -}} {{- $.Page.Scratch.SetInMap "cited_keys" $key true -}} {{- /* Use a clean fallback if title is missing */ -}} {{- $titleText := index $entry "title" | default "No Title" -}} {{- $tooltipText := printf "%s (%s). %s" $authorText $year $titleText -}} {{- /* Build HTML structure without nesting layouts inside the printf token evaluation loop */ -}} {{- $tooltipHTML := printf `%s` $tooltipText -}} {{- $citeHTML := "" -}} {{- if or (eq $style "n") (eq $style "narrative") -}} {{- $linkHTML := printf `%s` $key $authorText -}} {{- $citeHTML = printf `%s%s (%s)` $linkHTML $tooltipHTML $year -}} {{- else -}} {{- $linkHTML := printf `%s %s` $key $authorText $year -}} {{- $citeHTML = printf `%s%s` $linkHTML $tooltipHTML -}} {{- end -}} {{- $renderedCites = $renderedCites | append $citeHTML -}} {{- else -}} {{- $errorHTML := printf `[Missing Citation Data: %s]` $key -}} {{- $renderedCites = $renderedCites | append $errorHTML -}} {{- end -}} {{- end -}} {{- $joinedCites := delimit $renderedCites "; " -}} {{- if or (eq $style "p") (eq $style "parentheses") -}} ({{ $joinedCites | safeHTML }}) {{- else -}} {{ $joinedCites | safeHTML }} {{- end -}}
