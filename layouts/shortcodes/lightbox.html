{{- $src := .Get "src" -}} {{- $caption := .Get "caption" -}} {{- $gallery := .Get "gallery" -}} {{- $ratio := .Get "ratio" -}} {{- $showCaptionBelow := .Get "showCaptionBelow" -}} {{- $size := .Get "size" | default "130" -}} {{- $matchedResource := .Page.Resources.GetMatch $src -}} {{- if and (not $matchedResource) .Page.Parent -}} {{- $matchedResource = .Page.Parent.Resources.GetMatch $src -}} {{- end -}} {{- $image := "" -}} {{- if and $matchedResource (eq $matchedResource.ResourceType "image") -}} {{- $image = $matchedResource -}} {{- end -}} {{- $link := "" -}} {{- if $image -}} {{- $link = $image.RelPermalink -}} {{- else -}} {{- $link = $src | relURL -}} {{- end -}} {{- $thumbSrc := $link -}} {{- $thumbHeight := int $size -}} {{- if and $image $ratio -}} {{- $parts := split $ratio "/" -}} {{- if eq (len $parts) 2 -}} {{- $w := float (index $parts 0) -}} {{- $h := float (index $parts 1) -}} {{- $thumbWidth := int (math.Round (mul (div $w $h) $thumbHeight)) -}} {{- $cropSpec := printf "%dx%d Fill" $thumbWidth $thumbHeight -}} {{- $thumbSrc = ($image.Resize $cropSpec).RelPermalink -}} {{- end -}} {{- end -}} {{- $captionAttr := "" -}} {{- $cleanHTML := "" -}} {{- if $caption -}} {{- $cleanHTML = $caption | markdownify -}} {{- $cleanHTML = replace $cleanHTML "'" "'" -}} {{- $captionAttr = printf "data-caption='%s'" $cleanHTML | safeHTMLAttr -}} {{- end -}}
